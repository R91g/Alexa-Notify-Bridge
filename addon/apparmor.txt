#include <tunables/global>

profile alexa_notify_bridge flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Capabilities required for bootstrap and privilege drop (su-exec)
  capability setgid,
  capability setuid,
  file,
  signal (send) set=(kill,term,int,hup,cont),

  # Network
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,

  # Bootstrap / S6-Overlay / Bashio / su-exec
  /init ix,
  /run.sh ix,
  /bin/** ix,
  /sbin/su-exec ix,
  /usr/bin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/s6-overlay/** rwix,
  /run/{,**} rwk,
  /dev/tty rw,
  /usr/lib/bashio/** ix,
  /tmp/** rwk,
  /data/** r,

  # Transition the Python API server into a tightly confined child profile
  /usr/bin/python3* cx -> python_app,
  /usr/local/bin/python3* cx -> python_app,

  profile python_app flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>

    # Network (local HTTP listener + outbound HTTPS/DNS to Amazon)
    network inet stream,
    network inet6 stream,
    network inet dgram,
    network inet6 dgram,

    signal (receive) set=(kill,term,int,hup,cont),

    # Read-only application code and Python runtime/libraries
    /app/ r,
    /app/** r,
    /usr/ r,
    /usr/** mr,
    /lib/ r,
    /lib/** mr,
    /etc/ r,
    /etc/** r,
    /proc/ r,
    /proc/** r,
    /sys/ r,
    /sys/** r,

    # Standard devices and temporary files
    /dev/null rw,
    /dev/random r,
    /dev/urandom r,
    /tmp/ r,
    /tmp/** rwk,
  }
}
